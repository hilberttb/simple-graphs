###############################################################################
# Simple Graphs - snapshot scheduling
#
# We hook on_game_start_after_lobby rather than on_game_start. Both fire on the
# first in-game day, but vanilla documents on_game_start_after_lobby as running
# once the lobby is exited, i.e. after game rules have been applied. That
# matters here: several vanilla game rules rewrite county faiths at game start
# (see eliminate_&_replace_faith_effect), so snapshotting after the lobby gives
# a day-one count of the world the player is actually going to play.
#
# We do NOT append an "effect" block to the vanilla on_action - per
# common/on_action/_on_actions.info you cannot have two effect blocks on the
# same on_action. The sanctioned way to attach to a vanilla on_action without
# overriding the file is to have it call your own on_action, which is what this
# does.
###############################################################################

on_game_start_after_lobby = {
	on_actions = { stats_tracker_start }
}

stats_tracker_start = {
	events = { stats_tracker.1 }
}
