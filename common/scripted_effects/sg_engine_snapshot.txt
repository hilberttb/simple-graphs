###############################################################################
# Simple Graphs - ENGINE: the snapshot clock and the registry helper.
#
# ENGINE FILE. Subject to the two rules in README_GENERICITY_INVESTIGATION.md:
#   1. no sg_engine_* file names a game object type;
#   2. no sg_engine_* file references anything sg_c_*.
# Nothing below knows what is being measured or what kind of object is being
# measured. Every name a consumer owns arrives as a parameter.
#
# ---------------------------------------------------------------------------
# sg_engine_begin_snapshot_effect
# ---------------------------------------------------------------------------
# Advances the two global per-snapshot counters. MUST run before any consumer
# appends a point in the same snapshot, because both encodings in
# sg_engine_storage.txt read sg_snapshot_index as their uniquifying term.
#
#   sg_snapshot_index   1, 2, 3, ...   the snapshot NUMBER. This is the one the
#                                      renderer needs: it tags plain values and
#                                      is the high-order term of a packed
#                                      segment.
#   sg_snapshot_base    10000, 20000, ...  the older scaled counter, kept
#                                      because a consumer may still encode
#                                      base + value as a cross-check.
#   sg_snapshot_year    the ingame year this snapshot was taken in.
#
# Both counters are guarded on first use because neither global exists until
# the first snapshot runs. sg_snapshot_year is not, because it is overwritten
# rather than incremented.
#
# ---------------------------------------------------------------------------
# WHY THE YEAR IS READ FROM THE GAME RATHER THAN DERIVED FROM THE INDEX
# ---------------------------------------------------------------------------
# The x axis wants a real year per point. The obvious cheap alternative -
# store the start year once and compute start_year + sg_snapshot_index - is
# WRONG, and not marginally:
#
#   sg_snapshot_index advances on EVERY snapshot, including console-forced
#   ones. A save that has had a few hundred forced snapshots run against it
#   would report years centuries into the future while the underlying counts
#   are perfectly fine.
#
# Reading the actual date is immune to that, because it records what happened
# instead of assuming a cadence. Forced snapshots then simply produce several
# points sharing one year, which is the truth.
#
# `value = current_year` inside set_variable / set_global_variable is
# vanilla-proven, not inferred:
#   current_year            logs/triggers.log:556-558, "Compare the current
#                           ingame year", Supported Scopes: none
#   set_variable        value = current_year
#                           00_ep1_hold_court_effects.txt:34-37
#   set_global_variable value = current_year
#                           common/on_action/<the landed-realm on_actions file>:79-84
# and it is used as a value in ~20 further vanilla sites including
# common/script_values/.
#
# ---------------------------------------------------------------------------
# sg_engine_register_object_effect
# ---------------------------------------------------------------------------
#   sg_engine_register_object_effect = {
#       REGISTRY   = <global list name>   the list of objects to append to
#       VALUE_VAR  = <variable name>      this object's metric for this snapshot
#       MARKER_VAR = <variable name>      the sticky "has counted" marker
#   }
#
# Called in the scope of ONE object, while VALUE_VAR still holds this
# snapshot's number for it.
#
# WHAT IT IS FOR
# --------------
# The GUI needs some list to bind a datamodel to in order to answer "which
# objects exist, so a window can show or pick one". Script cannot build a
# variable name at runtime, so "one global list per object" was never possible.
# The working answer is the inverse: ONE list whose ENTRIES ARE THE OBJECTS.
#
# Storing the SCOPE rather than a key is the whole trick, and it is what makes
# this generic:
#   - objects created at runtime have engine-generated keys, so anything
#     key-based is fragile exactly where this needs to be robust;
#   - a stored scope needs no lookup at all. The GUI promotes it straight back
#     with datacontext = "[Scope.<Type>]" and can then read that object's OWN
#     per-object lists with <Type>.MakeScope.GetList('name')
#     (Scope.GetList, data_types_script.txt:3429).
#
# This is not a new 1.19 mechanism and does not need one. It is exactly what
# vanilla does in common/scripted_effects/03_dlc_fp2_scripted_effects.txt,
# lines 10-23 and 44-65: the same iteration, also storing the scope.
# $REGISTRY$ substitution into `name =` is likewise vanilla-proven
# (00_diarchy_scripted_effects.txt:60).
#
# THE FILTER
# ----------
# Registering every object would list hundreds of them, nearly all measuring
# zero, which makes a picker useless. So an object is registered once its
# metric has been above zero at least once, and the marker is STICKY: once set
# it is never cleared, so an object that later drops back to zero keeps its
# history reachable instead of vanishing from the UI mid-campaign.
#
# To register everything unconditionally instead, delete the has_variable limit
# on the second if-block. To register only objects above zero RIGHT NOW, change
# that limit to `var:$VALUE_VAR$ > 0` and drop the first if-block.
#
# Note this is a filter on WHAT GETS LISTED only. Collection is untouched:
# every object still gets its history appended every snapshot, whether or not
# it is registered.
#
# MARKER_VAR is a parameter rather than a fixed name because it PERSISTS
# between invocations, exactly like PREV_Y_VAR in sg_engine_storage.txt. Two
# consumers that could ever measure the same object must not share it.
###############################################################################

sg_engine_begin_snapshot_effect = {

	if = {
		limit = { NOT = { has_global_variable = sg_snapshot_base } }
		set_global_variable = { name = sg_snapshot_base value = 0 }
	}
	change_global_variable = { name = sg_snapshot_base add = 10000 }

	if = {
		limit = { NOT = { has_global_variable = sg_snapshot_index } }
		set_global_variable = { name = sg_snapshot_index value = 0 }
	}
	change_global_variable = { name = sg_snapshot_index add = 1 }

	# The ingame year this snapshot is being taken in. Read from the game, not
	# derived from the index - see the header for why that distinction matters.
	set_global_variable = { name = sg_snapshot_year value = current_year }
}

sg_engine_register_object_effect = {

	# Sticky marker: "this object's metric has been above zero at some point".
	# var:x > 0 in a limit is vanilla syntax (00_decisions_effects.txt:1537).
	if = {
		limit = { var:$VALUE_VAR$ > 0 }
		set_variable = { name = $MARKER_VAR$ value = 1 }
	}

	# save_scope_as then target = scope:... is copied from the vanilla fp2
	# effects above rather than using `target = this`, because that is the form
	# vanilla actually ships for this exact job. The scope name is reused every
	# iteration, which is fine and is also what vanilla does.
	if = {
		limit = { has_variable = $MARKER_VAR$ }
		save_scope_as = sg_registry_obj
		add_to_global_variable_list = {
			name = $REGISTRY$
			target = scope:sg_registry_obj
		}
	}
}

###############################################################################
# GROUPED REGISTRIES
#
# A flat registry answers "which objects exist". It does not answer "which
# objects belong together", and a picker listing a few hundred rows in raw
# iteration order is the readability problem those two effects exist to fix.
#
# The shape is one level of nesting and nothing more:
#
#   <group registry>          a global list whose entries are GROUP objects
#   <member list>             a per-group variable list, living ON the group
#                             object, whose entries are that group's MEMBERS
#
# and the GUI walks it as a datamodel inside a datamodel:
#
#   outer  datamodel = "[GetGlobalList( '<group registry>' )]"
#   inner  datamodel = "[<Type>.MakeScope.GetList( '<member list>' )]"
#
# Both halves are mechanisms this mod already ships and has verified in-game -
# the outer is exactly what sg_engine_register_object_effect feeds today, and
# the inner is the same Scope.GetList read-back (data_types_script.txt:3429)
# that every plot expression in the window already uses. Nothing new is being
# assumed about the engine; this is the existing registry pattern applied once
# more, one level down.
#
# WHY THE MEMBER LIST LIVES ON THE GROUP OBJECT
# ---------------------------------------------
# Script cannot build a variable name at runtime, so "one global list per
# group" is impossible - the same constraint that produced the registry
# pattern in the first place. Hanging the list off the group object sidesteps
# it completely: every group uses the SAME list name, because each one is a
# different object's variable.
#
# WHY MEMBERSHIP IS STORED RATHER THAN QUERIED IN THE GUI
# -------------------------------------------------------
# For some types the GUI can already enumerate a group's members on its own.
# Doing it that way would list EVERY member, including the ones this mod has
# never tracked, and a datamodel cannot be filtered - only each row's `visible`
# can, which leaves the group header showing above an empty body whenever a
# group has no tracked members at all.
#
# Storing the membership makes the filter structural instead: a group is only
# ever registered when it has a member to register, and a member list only ever
# contains members that passed the caller's own filter. The GUI then renders
# what it is given, with no visibility arithmetic anywhere.
#
# ---------------------------------------------------------------------------
# sg_engine_clear_group_registry_effect
# ---------------------------------------------------------------------------
#   sg_engine_clear_group_registry_effect = {
#       REGISTRY    = <global list name>
#       MEMBER_LIST = <per-group variable list name>
#   }
#
# Called in scope none, once per snapshot, BEFORE anything registers. Empties
# every member list belonging to a currently-registered group and then empties
# the registry itself, so both are rebuilt from scratch.
#
# ORDER IS LOAD-BEARING: the member lists have to be cleared while the registry
# still names their owners. Clearing the registry first would strand every one
# of them, and a stale member list is not harmless - it would show last
# snapshot's membership under this snapshot's headers.
#
# Rebuilding rather than appending is what makes objects created mid-campaign
# appear by themselves, for exactly the reason the flat registry is rebuilt
# too, and it makes de-duplication moot for the member lists.
#
# Both clears are guarded, matching the guard style of the flat registry's own
# clear sites: on the first snapshot neither list has ever existed, and whether
# clearing a never-created list is a silent no-op or an error.log line is not
# something this project has verified.
#
# ---------------------------------------------------------------------------
# sg_engine_register_group_member_effect
# ---------------------------------------------------------------------------
#   sg_engine_register_group_member_effect = {
#       REGISTRY    = <global list name>
#       MEMBER_LIST = <per-group variable list name>
#       MEMBER      = <a saved scope name>
#   }
#
# CALLED IN THE SCOPE OF THE GROUP OBJECT, not of the member. The member
# arrives as a saved scope because the two are different objects and only one
# of them can be `this`.
#
# Finding which group a member belongs to is the CALLER's job, and it has to
# be: it is the one part of this that cannot be written without naming a type.
# Some callers get the group from a scope link, others have to search the
# registry for it. By the time this effect runs, that question is answered and
# what is left is pure bookkeeping.
#
# The registry add is guarded by is_target_in_global_variable_list
# (vanilla: common/achievements/fp1_achievements.txt:6-9) because this effect
# is called once per MEMBER, so the second and every later member of a group
# would otherwise re-register the group it already registered. The member add
# is deliberately NOT guarded: a caller that hands the same member twice has a
# bug in its own iteration, and silently absorbing that would hide it.
###############################################################################

sg_engine_clear_group_registry_effect = {

	if = {
		limit = { has_global_variable_list = $REGISTRY$ }

		every_in_global_list = {
			variable = $REGISTRY$
			if = {
				limit = { has_variable_list = $MEMBER_LIST$ }
				clear_variable_list = $MEMBER_LIST$
			}
		}

		clear_global_variable_list = $REGISTRY$
	}
}

sg_engine_register_group_member_effect = {

	save_scope_as = sg_group_obj

	if = {
		limit = {
			OR = {
				NOT = { has_global_variable_list = $REGISTRY$ }
				NOT = {
					is_target_in_global_variable_list = {
						name = $REGISTRY$
						target = scope:sg_group_obj
					}
				}
			}
		}
		add_to_global_variable_list = {
			name = $REGISTRY$
			target = scope:sg_group_obj
		}
	}

	add_to_variable_list = {
		name = $MEMBER_LIST$
		target = scope:$MEMBER$
	}
}
