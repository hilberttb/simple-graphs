namespace = sg_engine

###############################################################################
# sg_engine.1 - the recurring snapshot. Fires the collection hook in
#               common/scripted_effects/sg_wiring.txt, which is the only
#               thing this file knows about any consumer.
#
# scope = none : this is world bookkeeping - there is no one it happens TO.
#                (documented in game/events/_events.info)
# hidden = yes : background event, no popup. _events.info also notes that
#                events with no personal scope generally need to be hidden.
#
# TIMING - why "years = 1" and not "days = 365":
#   _events.info documents days/weeks/months/years as trigger_event delays, so
#   years is a real, supported delay unit and the date arithmetic is left to
#   the engine instead of being open-coded.
#
#   The concern about a flat day-count drifting off the anniversary does not
#   apply to CK3 in any case: this engine has no leap years. The Jomini date
#   code (jomini_historical_date.cpp) documents its own DaysPerYear constant as
#   "Number of days per year, no leap years". A year is therefore always
#   exactly 365 days, and the snapshot lands on the same calendar day forever.
#   Vanilla history files do contain stray dates like 1132.2.29 and even
#   1002.2.30, but those are data artifacts, not evidence of a leap calendar.
###############################################################################

sg_engine.1 = {
	scope = none
	hidden = yes

	immediate = {
		sg_snapshot_collect_effect = yes

		# Re-arm for the same calendar day next year, indefinitely.
		trigger_event = {
			id = sg_engine.1
			years = 1
		}
	}
}
